📋 INSTALACIÓN RÁPIDA (MÉTODO DOCKER)
═══════════════════════════════════════════════════════════════


🔧 1. PREPARACIÓN INICIAL
───────────────────────────

# Conectar por SSH
ssh admin@tu-synology.local


# Crear directorios
sudo mkdir -p /volume1/docker/servyauto
sudo mkdir -p /volume1/docker/postgres


═══════════════════════════════════════════════════════════════


🗄️ 2. INSTALAR POSTGRESQL
──────────────────────────

docker run -d \
  --name postgres-servyauto \
  --restart always \
  -e POSTGRES_USER=servyauto \
  -e POSTGRES_PASSWORD=TuPasswordSeguro123 \
  -e POSTGRES_DB=servyauto_db \
  -v /volume1/docker/postgres:/var/lib/postgresql/data \
  -p 5432:5432 \
  postgres:14-alpine


# Verificar que está corriendo
docker ps | grep postgres


═══════════════════════════════════════════════════════════════


📂 3. SUBIR ARCHIVOS
────────────────────

# Opción A: Usando FileStation (más fácil)
1. Abrir FileStation en DSM
2. Ir a /docker/servyauto/
3. Subir y extraer servyauto_manager_v89_completo.zip


# Opción B: Usando SCP desde tu PC
scp servyauto_manager_v89_completo.zip admin@tu-synology:/volume1/docker/servyauto/

# Luego en Synology:
cd /volume1/docker/servyauto
unzip servyauto_manager_v89_completo.zip


═══════════════════════════════════════════════════════════════


⚙️ 4. CONFIGURAR .ENV
─────────────────────

cd /volume1/docker/servyauto/servyauto_manager_v89/app
nano .env


# Pegar este contenido (ajustar valores):
───────────────────────────────────────────────────────────────
DATABASE_URL="postgresql://servyauto:TuPasswordSeguro123@postgres-servyauto:5432/servyauto_db?schema=public"
NEXTAUTH_URL="http://tu-synology.local:3000"
NEXTAUTH_SECRET="resultado-de-comando-siguiente"
PORT=3000
───────────────────────────────────────────────────────────────


# Generar NEXTAUTH_SECRET
openssl rand -base64 32
# Copiar resultado y pegarlo en NEXTAUTH_SECRET


# Guardar archivo: Ctrl+X, Y, Enter


═══════════════════════════════════════════════════════════════


🐳 5. CREAR DOCKERFILE
──────────────────────

cd /volume1/docker/servyauto/servyauto_manager_v89
nano Dockerfile


# Pegar este contenido:
───────────────────────────────────────────────────────────────
FROM node:18-alpine

RUN apk add --no-cache libc6-compat openssl

WORKDIR /app

COPY app/package.json app/yarn.lock ./
RUN yarn install --frozen-lockfile

COPY app/ ./
RUN yarn prisma generate
RUN yarn build

EXPOSE 3000
ENV NODE_ENV=production

CMD ["yarn", "start"]
───────────────────────────────────────────────────────────────


# Guardar: Ctrl+X, Y, Enter


═══════════════════════════════════════════════════════════════


🔨 6. CONSTRUIR IMAGEN DOCKER
──────────────────────────────

cd /volume1/docker/servyauto/servyauto_manager_v89

docker build -t servyauto:latest .

# Esto toma 5-10 minutos
# Esperar hasta ver: "Successfully built..."


═══════════════════════════════════════════════════════════════


🚀 7. EJECUTAR APLICACIÓN
─────────────────────────

docker run -d \
  --name servyauto-app \
  --restart always \
  -p 3000:3000 \
  -v /volume1/docker/servyauto/servyauto_manager_v89/app/.env:/app/.env \
  --link postgres-servyauto:postgres \
  servyauto:latest


# Verificar estado
docker ps


# Ver logs en tiempo real
docker logs -f servyauto-app

# Presiona Ctrl+C para salir de los logs

🌐 10. CONFIGURAR ACCESO DESDE INTERNET (OPCIONAL)
───────────────────────────────────────────────────

A. Configurar Reverse Proxy:
   DSM → Control Panel → Login Portal → Advanced → Reverse Proxy
   
   Create:
   - Source Protocol: HTTPS
   - Source Hostname: servyauto.tudominio.com
   - Source Port: 443
   - Destination Protocol: HTTP
   - Destination Hostname: localhost
   - Destination Port: 3000


B. Configurar SSL:
   Control Panel → Security → Certificate
   Add → Let's Encrypt
   Domain: servyauto.tudominio.com


C. Port Forwarding en router:
   Puerto 443 → IP de Synology


D. Actualizar .env:
   nano /volume1/docker/servyauto/servyauto_manager_v89/app/.env
   
   Cambiar:
   NEXTAUTH_URL="https://servyauto.tudominio.com"
   
   Reiniciar:
   docker restart servyauto-app

❌ SOLUCIÓN DE PROBLEMAS
═════════════════════════


PROBLEMA: "Cannot find module @prisma/client"
SOLUCIÓN:
  docker exec servyauto-app yarn prisma generate
  docker restart servyauto-app


PROBLEMA: Error de conexión a base de datos
SOLUCIÓN:
  docker logs postgres-servyauto
  docker exec -it postgres-servyauto psql -U servyauto -d servyauto_db -c "SELECT 1;"


PROBLEMA: Puerto 3000 ya en uso
SOLUCIÓN:
  # Ver qué está usando el puerto
  sudo netstat -tuln | grep 3000
  
  # Detener proceso que usa el puerto
  docker ps
  docker stop <nombre-contenedor>
  
  # O cambiar puerto en docker run:
  -p 3001:3000


PROBLEMA: Contenedor no arranca
SOLUCIÓN:
  docker logs servyauto-app
  docker ps -a
  docker rm servyauto-app
  # Revisar .env y volver a ejecutar docker run

📊 SCRIPT DE BACKUP AUTOMÁTICO
═══════════════════════════════


# Crear archivo de backup
nano /volume1/docker/servyauto/backup.sh


# Contenido:
───────────────────────────────────────────────────────────────
#!/bin/bash
DATE=$(date +%Y%m%d_%H%M%S)
BACKUP_DIR="/volume1/docker/servyauto/backups"
mkdir -p $BACKUP_DIR

# Backup de base de datos
docker exec postgres-servyauto pg_dump -U servyauto servyauto_db > $BACKUP_DIR/db_$DATE.sql

# Comprimir
gzip $BACKUP_DIR/db_$DATE.sql

# Eliminar backups antiguos (más de 30 días)
find $BACKUP_DIR -name "*.sql.gz" -mtime +30 -delete

echo "Backup completado: db_$DATE.sql.gz"
───────────────────────────────────────────────────────────────


# Dar permisos
chmod +x /volume1/docker/servyauto/backup.sh


# Probar
/volume1/docker/servyauto/backup.sh


# Configurar en Task Scheduler (DSM):
# Control Panel → Task Scheduler → Create → Scheduled Task
# Script: /volume1/docker/servyauto/backup.sh
# Schedule: Daily at 2:00 AM
